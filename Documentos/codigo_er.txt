Table Usarios {
  id_usuario int [not null, primary key, increment]
  username varchar(20) [not null, unique]
  email varchar (255) [not null, unique]
  nombre varchar(50) [not null]
  apellidos varchar(50) [not null]
  rol varchar(9) [not null]
  passwd varchar(50) [not null]
  estado varchar(15) [not null]
  f_nac date [not null]
  sexo varchar(5) [not null]
}

Table Inquilinos {
  id_inquilino int [not null, primary key, increment]
  estudiante bool
  fuma bool
  mascota bool
  descripción varchar(500)
}

Table Caseros {
  id_casero int [not null, primary key, increment]
}

Table Viviendas {
  id_vivienda int [not null, pk, increment]
  id_casero int [not null, ref: > Caseros.id_casero]
  area int [not null]
  direccion varchar(50) [not null]
  max_inquilinos int [not null]
  descripcion varchar(500)
}

Table Anuncios {
  id_anuncio int [not null, pk, increment]
